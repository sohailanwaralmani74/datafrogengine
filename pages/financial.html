import { FinancialEngine } from '/assets/js/index.js';

// 1. Parse QBO / QFX or QIF
const statement = FinancialEngine.parse(qboString);

// 2. Auto-categorize transactions

  { pattern: 'Amazon', category: 'Office Supplies' },
  { pattern: 'Payroll', category: 'Income' }
]);

// 3. Reconcile against source records
const report = FinancialEngine.reconcile(statement, internalLedger);
