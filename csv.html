Example CSV Workflow

import { CsvEngine } from '/assets/js/index.js';

const rawCsv = `id,country,sales,year\n1,USA,50000,2025\n2,Germany,34000,2025\n3,USA,62000,2026`;

// 1. Reorder columns & filter rows
const filtered = CsvEngine.filter(rawCsv, row => Number(row.sales) > 40000);
const reordered = CsvEngine.reorderColumns(filtered, ['country', 'sales', 'year']);

// 2. Group by country with aggregation
const summary = CsvEngine.groupBy(rawCsv, 'country', {
  totalSales: rows => rows.reduce((acc, r) => acc + Number(r.sales), 0)
});

// 3. Convert to Excel or JSON
const json = CsvEngine.toJson(reordered);
console.log('Summary JSON:', json);
