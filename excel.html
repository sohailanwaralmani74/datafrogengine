Example Excel Workflow

import { ExcelEngine } from '/assets/js/index.js';

// 1. Build an Excel workbook fluently
const buffer = await ExcelEngine.builder()
  .addSheet('Financial Summary')
  .addRow(['Department', 'Q1 Budget', 'Q1 Spend'])
  .addRow(['Engineering', 150000, 142000])
  .addRow(['Marketing', 80000, 78500])
  .build();

// 2. Convert directly to CSV or JSON
const csv = ExcelEngine.toCsv(buffer);
const json = ExcelEngine.toJson(buffer);

// 3. Audit cell metrics
const analysis = ExcelEngine.analyze(buffer);
console.log('Total sheets:', analysis.sheetCount);
