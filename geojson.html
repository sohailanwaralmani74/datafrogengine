Example GeoJSON Workflow

import { GeoJsonEngine } from '/assets/js/index.js';

const point = {
  type: 'Feature',
  geometry: { type: 'Point', coordinates: [-122.4194, 37.7749] },
  properties: { city: 'San Francisco' }
};

// 1. Validate RFC 7946 bounds
const isValid = GeoJsonEngine.isValid(point);

// 2. Convert to WKT or KML
const wkt = GeoJsonEngine.toWkt(point);
const kml = GeoJsonEngine.toKml(point);

// 3. Calculate Bounding Box
const bbox = GeoJsonEngine.calculateBBox(point);
console.log('WKT representation:', wkt);
