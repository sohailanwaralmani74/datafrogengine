Interactive Code Example

import { PdfEngine } from '/assets/js/index.js';

// 1. Load an existing PDF document
const pdfDoc = await PdfEngine.load(pdfBuffer);

// 2. Reorder pages and rotate orientation
const reordered = PdfEngine.reorderPages(pdfBuffer, [2, 0, 1]);
const rotated = PdfEngine.rotateAll(reordered, 90);

// 3. Add diagonal watermark and anonymize
const watermarked = PdfEngine.addWatermark(rotated, 'CONFIDENTIAL');
const sanitized = PdfEngine.anonymize(watermarked);

// 4. Split or extract text
const text = PdfEngine.toText(sanitized);
console.log('Extracted text:', text);
