Example Data Workflow

import { JsonTool } from '/assets/js/index.js';

const raw = `{ status: "ok", count: 12, details: { tags: ["prod", "us-east"], } }`;

// 1. Repair malformed JSON
const repaired = JsonTool.repair(raw);
const parsed = JsonTool.parse(repaired);

// 2. Query with JSONPath
const tags = JsonTool.query(parsed, '$.details.tags[*]');

// 3. Infer Schema & Diff
const schema = JsonTool.inferSchema(parsed);
const diff = JsonTool.diff({ count: 10 }, parsed);
console.log('Differences:', diff.differences);
