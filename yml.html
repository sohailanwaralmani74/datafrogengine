Interactive Code Example

import { YmlEngine } from '/assets/js/index.js';

const yaml = `service: Datafrog\nversion: 2.1\nenv:\n  - production\n  - staging`;

// 1. Parse and deep mutate
const parsed = YmlEngine.parse(yaml);
YmlEngine.set(parsed, 'env.2', 'sandbox');

// 2. Key reordering and sorting
const sorted = YmlEngine.sortKeys(parsed);

// 3. Deep diff against source
const diff = YmlEngine.diff(yaml, sorted);
console.log('Diff summary:', diff.summary);
