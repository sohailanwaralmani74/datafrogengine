Example XML Workflow

import { XmlTool } from '/assets/js/index.js';

const xml = `<catalog><book id="bk101"><title>Datafrog Manual</title><price>44.95</price></book></catalog>`;

// 1. Parse into DOM tree
const doc = XmlTool.parse(xml);

// 2. Query nodes
const book = XmlTool.query(doc, 'book');
XmlTool.setAttribute(book, 'status', 'active');

// 3. Convert to JSON
const json = XmlTool.toJson(xml);
console.log('Converted JSON:', json);
